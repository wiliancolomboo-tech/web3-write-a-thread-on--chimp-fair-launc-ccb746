import "dotenv/config";
import express from "express";
import { Connection, PublicKey } from "@solana/web3.js";
import crypto from "node:crypto";

const RPC_URL = process.env.RPC_URL ?? "https://api.devnet.solana.com";
const LAUNCH = process.env.LAUNCH_ADDRESS;
const RESEND_API_KEY = process.env.RESEND_API_KEY;
const ALERT_WEBHOOK_URL = process.env.ALERT_WEBHOOK_URL;
const PORT = Number(process.env.PORT ?? 8080);

if (!LAUNCH) throw new Error("LAUNCH_ADDRESS is required");

const connection = new Connection(RPC_URL, "confirmed");
const launchKey = new PublicKey(LAUNCH);

async function sendAlert(message: string): Promise<void> {
  if (!ALERT_WEBHOOK_URL) return;

  const response = await fetch(ALERT_WEBHOOK_URL, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ content: message, text: message }),
  });

  if (!response.ok) {
    throw new Error(`Webhook failed: ${response.status}`);
  }
}

async function sendEmail(to: string, subject: string, html: string): Promise<void> {
  if (!RESEND_API_KEY) return;

  const response = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: {
      authorization: `Bearer ${RESEND_API_KEY}`,
      "content-type": "application/json",
    },
    body: JSON.stringify({
      from: process.env.EMAIL_FROM ?? "CHIMP Fair Launch <noreply@example.com>",
      to,
      subject,
      html,
    }),
  });

  if (!response.ok) {
    throw new Error(`Resend failed: ${response.status}`);
  }
}

async function getHealth() {
  const slot = await connection.getSlot("confirmed");
  const balance = await connection.getBalance(launchKey, "confirmed");

  return {
    ok: true,
    rpc: RPC_URL,
    launch: launchKey.toBase58(),
    slot,
    launchLamports: balance,
    timestamp: new Date().toISOString(),
  };
}

const app = express();
app.use(express.json());

app.get("/healthz", async (_req, res) => {
  try {
    res.status(200).json(await getHealth());
  } catch (error) {
    res.status(503).json({
      ok: false,
      error: error instanceof Error ? error.message : "unknown error",
    });
  }
});

app.get("/readyz", async (_req, res) => {
  try {
    await connection.getLatestBlockhash("confirmed");
    res.status(200).json({ ready: true });
  } catch {
    res.status(503).json({ ready: false });
  }
});

app.post("/notify", async (req, res) => {
  const { email, subject, html, message } = req.body ?? {};

  if (typeof email === "string" && typeof subject === "string" && typeof html === "string") {
    await sendEmail(email, subject, html);
  }

  if (typeof message === "string") {
    await sendAlert(message);
  }

  res.status(202).json({ accepted: true });
});

app.listen(PORT, () => {
  console.log(`CHIMP operations service listening on :${PORT}`);
});
